function tone(freq,dur,type="sine",gain=.035,when=0,slideTo=null){
 try{const c=ctx(),o=c.createOscillator(),g=c.createGain();o.type=type;o.frequency.setValueAtTime(freq,c.currentTime+when);if(slideTo)o.frequency.exponentialRampToValueAtTime(slideTo,c.currentTime+when+dur);g.gain.setValueAtTime(.0001,c.currentTime+when);g.gain.exponentialRampToValueAtTime(gain,c.currentTime+when+.008);g.gain.exponentialRampToValueAtTime(.0001,c.currentTime+when+dur);o.connect(g);g.connect(c.destination);o.start(c.currentTime+when);o.stop(c.currentTime+when+dur+.02)}catch{}}
